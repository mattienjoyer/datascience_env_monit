# Function to load and clean data from ZIPs
load_flux_data <- function(site_id) {
  zip_path <- list.files(here::here("data", "raw"), 
                         pattern = paste0("_", site_id, "_FLUXNET_.*\\.zip$"), 
                         full.names = TRUE)
  if(length(zip_path) == 0) return(NULL)
  
  zz <- unzip(zip_path[1], list = TRUE)
  hh_file <- zz$Name[grep("FLUXMET_HH", zz$Name)]
  
  df <- read.csv(unzip(zip_path[1], files = hh_file), 
                 na.strings = "-9999", 
                 colClasses = c(TIMESTAMP_START = "character"))
  
  available_vars <- intersect(colnames(df), KEEP_VARS)
  df <- df[, available_vars]
  
  df <- df %>% 
    mutate(time = as.POSIXct(TIMESTAMP_START, format = "%Y%m%d%H%M", tz = "UTC"), 
           site = site_id) %>% 
    select(-TIMESTAMP_START)
  return(df)
}

# General statistics – ensure sentinel "-9999" is treated as missing
stats_long <- all_data %>%
  select(site, all_of(names(vars_labels))) %>%
  # Coerce to numeric (in case columns were read as character) and replace -9999 with NA
  mutate(across(all_of(names(vars_labels)), ~as.numeric(.x))) %>%
  mutate(across(all_of(names(vars_labels)), ~na_if(.x, -9999))) %>%
  pivot_longer(-site, names_to = "var") %>%
  summarise(
    Mean = mean(value, na.rm = TRUE),
    Min = min(value, na.rm = TRUE),
    Max = max(value, na.rm = TRUE),
    `% missing` = 100 * mean(is.na(value)),
    .by = c(site, var)
  )

# Annual precipitation mean sum
p_annual <- all_data %>%
  mutate(year = year(time)) %>%
  summarise(p_sum = sum(P_F, na.rm = TRUE), .by = c(site, year)) %>%
  summarise(p_annual = mean(p_sum), .by = site)

stats_final <- stats_long %>%
  left_join(p_annual, by = "site") %>%
  mutate(Mean = if_else(var == "P_F", p_annual, Mean)) %>%
  select(-p_annual)

# Tables per site – render as HTML tables with styling

    # Prepare daily fluxes, convert units and clean unrealistic RECO values for DE‑HoH
    daily_flux <- all_data %>%
      # Convert sentinel -9999 to NA for the three flux columns before any averaging
      mutate(across(c(NEE_VUT_REF, GPP_NT_VUT_REF, RECO_NT_VUT_REF), ~na_if(.x, -9999))) %>%
      mutate(date = as.Date(time)) %>%
      group_by(site, date) %>%
      summarise(
        across(c(NEE_VUT_REF, GPP_NT_VUT_REF, RECO_NT_VUT_REF),
                ~mean(.x, na.rm = TRUE) * 1.0378),

      ) %>%
  pivot_longer(cols = c(NEE_VUT_REF, GPP_NT_VUT_REF, RECO_NT_VUT_REF),
               names_to = "variable", values_to = "value") %>%

                            NEE_VUT_REF = "NEE",
                            GPP_NT_VUT_REF = "GPP",
                            RECO_NT_VUT_REF = "RECO")) %>%
  # Remove extreme RECO outliers for DE‑HoH (abs > 5000)

eb <- all_data %>%
  # ensure missing flags are true NA
  mutate(across(c(NETRAD, G_F_MDS, H_F_MDS, LE_F_MDS), ~na_if(.x, -9999))) %>%
  filter(site == "DE-HoH", H_F_MDS_QC == 0, LE_F_MDS_QC == 0) %>%

The strongest carbon uptake (most negative mean annual NEE) is observed at the DE‑HoH tower (‑254 g C m⁻² yr⁻¹). This tower also has the longest record (12 complete years) and a relatively high share of gap‑filled NEE values (≈ 69 %). The high uptake is likely driven by the temperate forest ecosystem, which efficiently absorbs CO₂ during the growing season.

The AU‑Sam site shows a substantial uptake (‑128 g C m⁻² yr⁻¹) despite a shorter record (8 years) and a similar gap‑filled proportion (≈ 68 %). Its moderate GPP seasonal amplitude (≈ 5 g C m⁻² d⁻¹) suggests a relatively stable photosynthetic period.

CA‑Man exhibits the weakest uptake among the negative sites (‑22 g C m⁻² yr⁻¹) with 15 complete years and a comparable gap‑filled share (≈ 69 %). The low GPP amplitude (≈ 4.8 g C m⁻² d⁻¹) indicates limited seasonal variation, which may reflect the Mediterranean climate of the site.

The KR‑CRK tower is close to carbon neutral (≈ +1 g C m⁻² yr⁻¹) and shows the largest GPP seasonal amplitude (≈ 10.8 g C m⁻² d⁻¹), implying strong seasonal productivity swings typical for a temperate monsoon region.

Finally, US‑ICh is a net carbon source (+23 g C m⁻² yr⁻¹) even though it has the longest record (19 years). Its high gap‑filled NEE share (≈ 78 %) and the lowest GPP amplitude (≈ 3 g C m⁻² d⁻¹) suggest that measurement gaps and a relatively flat productivity curve limit carbon uptake.

Across all sites, the share of gap‑filled NEE values ranges from ~ 64 % (KR‑CRK) to ~ 78 % (US‑ICh), highlighting the importance of data completeness when interpreting the carbon balance.
